def search_papers(query: str, max_results: int = 5) -> str:
    words = [w for w in re.findall(r"[a-z]+", query.lower()) if len(w) > 2]
    hits = sorted(((sum(w in (p["title"] + " " + p["summary"]).lower() for w in words), pid)
                   for pid, p in PAPERS.items()), reverse=True)
    return json.dumps([{"id": pid, "title": PAPERS[pid]["title"]} for score, pid in hits[:max_results] if score])

def get_paper(paper_id: str) -> str:
    if paper_id not in PAPERS:
        return json.dumps({"error": f"unknown id {paper_id}; call search_papers first"})
    return json.dumps({"id": paper_id, **PAPERS[paper_id]})

def run_agent(system, task, tools, max_steps=8):
    """The simple loop from Lab 1."""
    msgs = [{"role": "system", "content": system}, {"role": "user", "content": task}]
    for _ in range(max_steps):
        r = client.chat.completions.create(model=MODEL, temperature=0, extra_body=EXTRA, messages=msgs,
                                           tools=[SPECS[t] for t in tools] or None)
        COST["chat"] += r.usage.cost
        msg = r.choices[0].message
        if not msg.tool_calls:
            return msg.content
        msgs.append(msg.model_dump(exclude_none=True))
        for call in msg.tool_calls:
            args = json.loads(call.function.arguments)
            print(f"  🔧 {call.function.name}({args})")
            msgs.append({"role": "tool", "tool_call_id": call.id, "content": FUNCS[call.function.name](**args)})
    return None

def ask_json(system, user, schema, name):
    """One model call with strict structured output (EP02); returns the parsed JSON.
    max_tokens: strict JSON can loop on whitespace until the limit (EP02) — a cut-off answer raises JSONDecodeError."""
    r = client.chat.completions.create(model=MODEL, temperature=0, extra_body=EXTRA, max_tokens=2000, messages=[
            {"role": "system", "content": system}, {"role": "user", "content": user}],
        response_format={"type": "json_schema", "json_schema": {"name": name, "strict": True, "schema": schema}})
    COST["chat"] += r.usage.cost
    return json.loads(r.choices[0].message.content)

Mem0 focuses on creating scalable long-term memory by having an LLM extract, update, or delete specific facts from conversations [2504.19413]. A-MEM approaches memory by organizing information into linked notes that use keywords and tags to connect new information to old data [2502.12110]. MemGPT treats memory management like a computer operating system, moving facts between a limited main context and a larger external storage area [2310.08560]. While Mem0 and A-MEM focus on how information is structured and updated, MemGPT focuses on the technical movement of data between different storage layers.

import json
from collections import Counter

def run(papers):
    """
    Calculates the number of papers published per year.
    
    Args:
        papers (dict): A dictionary where keys are arXiv IDs and values are 
                       dicts containing {'title', 'authors', 'year', 'summary'}.
    
    Returns:
        dict: A dictionary mapping years (as strings) to the count of papers.
    """
    years = []
    for paper_id in papers:
        year = papers[paper_id].get('year')
        if year is not None:
            # Ensure year is treated as a string for consistent JSON serialization
            years.append(str(year))
    
    counts = Counter(years)
    # Return sorted by year for better readability
    return dict(sorted(counts.items()))

import json

def run(papers):
    """
    Sorts the papers dictionary by year in descending order (newest to oldest).
    Returns a list of paper dictionaries sorted by year.
    """
    # Convert dict to list of dicts and include the id for reference
    paper_list = []
    for paper_id, info in papers.items():
        item = info.copy()
        item['id'] = paper_id
        paper_list.append(item)

    # Sort by year (descending). Handle cases where year might be missing or non-integer.
    def sort_key(x):
        try:
            return int(x.get('year', 0))
        except (ValueError, TypeError):
            return 0

    sorted_list = sorted(paper_list, key=sort_key, reverse=True)
    
    return sorted_list
