0. Setup

เราห่อการเรียก LLM ไว้ใน ask_llm() ฟังก์ชันเดียว ทุก pattern ด้านล่างใช้ฟังก์ชันนี้

  • OpenRouter ใช้ API แบบ OpenAI-compatible จึงใช้ openai SDK ได้เลย
  • reasoning — Gemma 4 ส่งความคิด (reasoning) กลับมาด้วย เราใช้เป็น scratchpad (Thought) ของ ReAct
  • require_parameters — บังคับให้ OpenRouter ส่ง request ไปเฉพาะ provider ที่รองรับ tools และ JSON schema
  • ทุก call บันทึกจำนวน token และค่าใช้จ่ายไว้ใน USAGE

client = OpenAI(base_url="https://openrouter.ai/api/v1", api_key=load_key())
MODEL = "google/gemma-4-26b-a4b-it"   # ลองเปลี่ยนเป็น model อื่นบน OpenRouter ได้
EXTRA = {"provider": {"require_parameters": True}, "reasoning": {"effort": "medium"}}

USAGE = {"llm_calls": 0, "input_tokens": 0, "output_tokens": 0, "cost_usd": 0.0}

def ask_llm(messages, tools=None, system=None, response_format=None):
    if system:
        messages = [{"role": "system", "content": system}] + messages
    kwargs = dict(model=MODEL, messages=messages, extra_body=EXTRA)
    if tools:
        kwargs["tools"] = tools
    if response_format:
        kwargs["response_format"] = response_format
    resp = client.chat.completions.create(**kwargs)
    USAGE["llm_calls"] += 1
    USAGE["input_tokens"] += resp.usage.prompt_tokens
    USAGE["output_tokens"] += resp.usage.completion_tokens
    USAGE["cost_usd"] += getattr(resp.usage, "cost", 0) or 0
    return resp.choices[0].message

print("ready:", MODEL)

0.1 Tool registry

Tool แต่ละตัวคือ Python function + JSON schema ที่บอก model ว่าเรียกอย่างไร ตำราเน้นว่า “Provide good tools” (§19.3) — description ชัดเจนและ schema เข้มงวด (additionalProperties: False) ช่วยให้ model เลือกใช้ tool ถูกต้อง

TOOLS = {}

def tool(description, properties, required=None):
    """Decorator: ลงทะเบียน function เป็น tool พร้อม schema"""
    def wrap(fn):
        TOOLS[fn.__name__] = {
            "fn": fn,
            "schema": {"type": "function", "function": {
                "name": fn.__name__,
                "description": description,
                "parameters": {
                    "type": "object",
                    "properties": properties,
                    "required": required or list(properties),
                    "additionalProperties": False,
                },
            }},
        }
        return fn
    return wrap

def schemas(names):
    return [TOOLS[n]["schema"] for n in names]

# ---- simulated paper corpus (ข้อมูลจำลองเพื่อการสอน) ----
CORE = [

# จำลองสถานการณ์ในตำรา: การค้นหาครั้งแรกเจอแค่ 3 ฉบับ ครั้งถัดไป (ค้นแบบกว้างขึ้น) จึงเจอเพิ่ม
@tool("Search academic papers on a topic. Returns a JSON list of {id, title, year}.",
      {"query": {"type": "string"}})
def search_papers(query):
    STATE["search_calls"] += 1
    hits = CORE if STATE["search_calls"] == 1 else CORE + ADJACENT
    STATE["papers_found"] = hits
    return json.dumps([{k: p[k] for k in ("id", "title", "year")} for p in hits])

@tool("Read papers by id (e.g. 'P1') and return their key methods.",
      {"paper_ids": {"type": "array", "items": {"type": "string"}}})
def read_papers(paper_ids):
    allp = {p["id"]: p for p in CORE + ADJACENT}
    unknown = [i for i in paper_ids if i not in allp]
    if unknown:   # error ที่ชัดเจนช่วยให้ model แก้ตัวเองได้ (Provide good tools, §19.3)
        return f"Error: unknown paper ids {unknown}. Use the ids returned by search_papers, e.g. 'P1'."
    return json.dumps([{"title": allp[i]["title"], "key_method": allp[i]["note"]} for i in paper_ids])

@tool("Evaluate an arithmetic expression, e.g. '(3+4)*2'.", {"expression": {"type": "string"}})
def calculator(expression):
    if not set(expression) <= set("0123456789+-*/(). "):
        return "Error: only numbers and + - * / ( ) are allowed"
    return str(eval(expression))

print("tools:", list(TOOLS))

Part 1 — ReAct loop (§19.2.1)

\[\text{Thought}_t \rightarrow \text{Action}_t \rightarrow \text{Observation}_t \rightarrow \text{Thought}_{t+1} \rightarrow \cdots\]

ตำราระบุ 4 สิ่งที่ harness ต้องมี — ดูคอมเมนต์ ①–④ ในโค้ด

  1. Scratchpad — Thought ถูก log แต่ไม่แสดงให้ผู้ใช้
  2. Tool parsing — ดึง tool call ที่มีโครงสร้างจาก output
  3. Max iterations — จำกัดจำนวนรอบเสมอ (10–25)
  4. Termination — จบเมื่อไม่มี tool call แล้ว
def run_tool(call):
    "Execute one tool call -> tool message (error ก็ต้องส่งกลับ ห้ามทิ้ง)"
    try:
        args = json.loads(call.function.arguments or "{}")
        out = TOOLS[call.function.name]["fn"](**args)
    except Exception as e:
        out = f"Error: {type(e).__name__}: {e}"
    return {"role": "tool", "tool_call_id": call.id, "content": str(out)}

def react_agent(task, tool_names, system=None, max_iterations=10, verbose=True, indent=""):
    messages = [{"role": "user", "content": task}]
    for i in range(1, max_iterations + 1):               # ③ max iterations
        msg = ask_llm(messages, tools=schemas(tool_names), system=system)
        thought = getattr(msg, "reasoning", None)          # ① scratchpad (log only)
        if verbose and thought:
            print(f"{indent}💭 [{i}] {' '.join(thought.split())[:300]}")

        calls = msg.tool_calls or []                       # ② tool parsing
        if not calls:                                      # ④ termination
            return msg.content
        messages.append({"role": "assistant", "content": msg.content or "",
                         "tool_calls": [c.model_dump() for c in calls]})
        for c in calls:
            if verbose:
                print(f"{indent}🔧 [{i}] {c.function.name}({c.function.arguments})")
        # หลาย tool call ใน response เดียว -> รันพร้อมกัน
        t0 = time.time()
        with ThreadPoolExecutor() as pool:
            results = list(pool.map(run_tool, calls))
        if verbose and len(calls) > 1:
            print(f"{indent}⏱️ [{i}] {len(calls)} tools ran in parallel: {time.time() - t0:.1f}s")
        for r in results:
            if verbose:
                print(f"{indent}👀 [{i}] {r['content'][:200]}")
        messages.extend(results)
    return "⚠️ Stopped: reached max_iterations"

Part 2 — Planning agent: Plan-and-Execute + adaptive replanning (§19.2.2, §18.5.2)

\[\text{Plan} = \text{Planner}(q), \qquad \text{Result} = \prod_{i=1}^{|\text{Plan}|} \text{Executor}(\text{Plan}[i], \text{context}_i)\]

ขั้นตอน: 1. Planning phase — LLM call เดียว สร้าง plan เป็น DAG (แต่ละ step มี deps) 2. Execution phase — รันเฉพาะ step ที่ deps เสร็จหมดแล้ว แต่ละ step ใช้ ReAct agent จาก Part 1 (pattern ประกอบกันได้!) 3. Plan revision — ถ้าผลลัพธ์ไม่เป็นไปตามคาด ให้ replan จากสถานะปัจจุบัน

เราใช้ structured output (JSON schema) ให้ plan เป็น JSON ตามโครงสร้างที่กำหนดเสมอ แล้ว validate ซ้ำด้วย Pydantic — ตรงกับหลัก “Use structured outputs” (§19.3)

from pydantic import BaseModel

class Step(BaseModel):
    id: int
    task: str
    tool: str          # "search_papers" | "read_papers" | "none"
    deps: list[int]

class Plan(BaseModel):
    steps: list[Step]

# JSON schema ที่ส่งให้ model — บังคับให้ตอบเป็น plan ที่มีโครงสร้างเสมอ
PLAN_FORMAT = {"type": "json_schema", "json_schema": {"name": "plan", "strict": True, "schema": {
    "type": "object", "additionalProperties": False, "required": ["steps"],
    "properties": {"steps": {"type": "array", "items": {
        "type": "object", "additionalProperties": False, "required": ["id", "task", "tool", "deps"],
        "properties": {
            "id": {"type": "integer"},
            "task": {"type": "string"},
            "tool": {"type": "string", "enum": ["search_papers", "read_papers", "none"]},
            "deps": {"type": "array", "items": {"type": "integer"}},
        }}}}}}}

def ask_plan(prompt):
    msg = ask_llm([{"role": "user", "content": prompt}], system=PLANNER_SYSTEM, response_format=PLAN_FORMAT)
    return validate_dag(Plan.model_validate_json(msg.content))

PLANNER_SYSTEM = (
    "You are a planner. Break the user's request into 4-7 steps forming a DAG.\n"
    "Each step has: id, task, tool (search_papers | read_papers | none), and deps.\n"
    "deps = ids of the earlier steps whose OUTPUT this step needs. Rules:\n"
    "- Only the first step(s) may have empty deps.\n"
    "- A step that reads, writes or reviews something must depend on the step that produced it.\n"
    "- Two steps that do not need each other's output must not depend on each other "
    "(so they can run in parallel)."
)

def validate_dag(plan):
    """ตรวจว่า deps อ้างถึง step ที่มีอยู่ และไม่มี cycle"""
    ids = {s.id for s in plan.steps}
    for s in plan.steps:
        missing = set(s.deps) - ids
        assert not missing, f"step {s.id} depends on unknown steps {missing}"
    done, remaining = set(), {s.id: set(s.deps) for s in plan.steps}
    while remaining:
        ready = [i for i, d in remaining.items() if d <= done]
        assert ready, f"cycle detected among steps {sorted(remaining)}"
        for i in ready:
            done.add(i); remaining.pop(i)
    return plan

def show_plan(plan, results=None):
    results = results or {}
    for s in plan.steps:
        mark = "✅" if s.id in results else "⬜"
        print(f"  {mark} {s.id}. {s.task}  [tool={s.tool}, deps={s.deps}]")

request = "Write a 2-page report comparing transformer architectures for time-series forecasting."
plan = ask_plan(request)
print("📋 PLAN"); show_plan(plan)

Executor + adaptive replanning

กลยุทธ์ที่ใช้คือ Adaptive (ตาราง 19.1) — replan เฉพาะเมื่อ step ไม่ผ่านเงื่อนไข

เงื่อนไขในตัวอย่าง: รายงานเปรียบเทียบต้องมี paper อย่างน้อย 5 ฉบับ ถ้าค้นเจอน้อยกว่านั้น → replan

MIN_PAPERS = 5

def check_step(step):
    """คืนเหตุผลที่ต้อง replan หรือ None ถ้าผ่าน"""
    if step.tool == "search_papers" and len(STATE["papers_found"]) < MIN_PAPERS:
        return (f"Step {step.id} found only {len(STATE['papers_found'])} papers; "
                f"the report needs at least {MIN_PAPERS}.")
    return None

def replan(request, plan, results, reason):
    done = {i: results[i][:300] for i in results}
    prompt = (
        f"REPLAN. Original request: {request}\n"
        f"Current plan: {plan.model_dump_json()}\n"
        f"Completed step outputs: {json.dumps(done, ensure_ascii=False)}\n"
        f"Problem: {reason}\n"
        "Return the full revised plan. Keep completed steps with the same ids and tasks. "
        "Add new step(s) with new ids to fix the problem and update deps of later steps."
    )
    return ask_plan(prompt)

def run_step(step, plan, results):
    context = {d: results[d] for d in step.deps}
    task = (f"Overall goal: {request}\nYour step: {step.task}\n"
            f"Outputs of prerequisite steps: {json.dumps(context, ensure_ascii=False)}\n"
            "Do only this step. Be concise (under 200 words). "
            "Keep any ids from tool results (e.g. paper ids like P1) in your output "
            "so that later steps can use them.")
    tools = [step.tool] if step.tool in TOOLS else []
    if tools:
        return react_agent(task, tools, max_iterations=6, indent="      ")
    return ask_llm([{"role": "user", "content": task}]).content

def execute_plan(request, plan, max_replans=2):
    results, replans = {}, 0
    while len(results) < len(plan.steps):
        ready = [s for s in plan.steps
                 if s.id not in results and all(d in results for d in s.deps)]
        print(f"\n▶️  ready to run: {[s.id for s in ready]}")
        for step in ready:
            print(f"   ⚙️  step {step.id}: {step.task}")
            out = run_step(step, plan, results)
            reason = check_step(step)
            if reason and replans < max_replans:
                print(f"   ⚠️  {reason} → REPLAN")
                results[step.id] = out
                plan = replan(request, plan, results, reason)
                replans += 1
                print("📋 REVISED PLAN"); show_plan(plan, results)
                break                      # คำนวณ ready ใหม่จาก plan ใหม่
            results[step.id] = out
    return plan, results

STATE.update(papers_found=[], search_calls=0)   # เริ่มฐานข้อมูลจำลองใหม่
final_plan, results = execute_plan(request, plan)
print("\n📄 FINAL OUTPUT (last step)\n", results[final_plan.steps[-1].id])

HIDDEN_TESTS = [            # evaluator's tests — agent ไม่เห็น
    ("1h30m", 90), ("2h", 120), ("45m", 45),
    ("90s", 1.5),           # seconds
    ("1H 15M", 75),         # uppercase + spaces
    ("1.5h", 90),           # decimals
    ("1d", 1440),           # days
    ("abc", ValueError),    # invalid input must raise ValueError
    ("10", ValueError),     # number without unit is ambiguous -> ValueError
]

def extract_code(text):
    return text.split("```python")[-1].split("```")[0] if "```" in text else text

def evaluate(code_str):
    ns = {}
    try:
        exec(code_str, ns)
        fn = ns["parse_duration"]
    except Exception as e:
        return [f"code does not run: {e}"]
    failures = []
    for inp, want in HIDDEN_TESTS:
        expects_error = isinstance(want, type) and issubclass(want, Exception)
        try:
            got = fn(inp)
            if expects_error:
                failures.append(f"parse_duration({inp!r}) returned {got}, expected {want.__name__}")
            elif abs(got - want) > 1e-9:
                failures.append(f"parse_duration({inp!r}) returned {got}, expected {want}")
        except Exception as e:
            if not (expects_error and isinstance(e, want)):
                failures.append(f"parse_duration({inp!r}) raised {type(e).__name__}: {e}")
    return failures

def reflexion(max_attempts=4):

4.4 Fallback (graceful degradation)

Harness ลอง tool ตามลำดับความสำคัญ — model ไม่ต้องรู้เรื่อง fallback เลย (§19.2.4, หลัก “Plan for failure” §19.3)

def primary_search(q):  raise TimeoutError("primary API timed out")
def backup_search(q):   raise RuntimeError("429 rate limited")
def cached_search(q):   return json.dumps([{"title": "PatchTST", "source": "cache (2 days old)"}])

def search_with_fallback(query):
    for name, fn in [("primary", primary_search), ("backup", backup_search), ("cache", cached_search)]:
        try:
            result = fn(query)
            print(f"   ✅ {name} succeeded")
            return result
        except Exception as e:
            print(f"   ❌ {name} failed: {e}")
    return "Search is currently unavailable."   # สุดท้าย: แจ้ง model ตรงๆ
